# Builds the same sources as ../ against the Beat Saber 1.40.8 libraries
# (bs-cordl 4008, beatsaber-hook 6, BSML 0.4, SongCore 1.1). Code that
# differs between the two game versions checks SBSC_BS_1_40.

# include some defines automatically made by qpm
include(qpm_defines.cmake)

# QPM derives MOD_ID from the display name. Preserve the installed identity
# and saved-file names when rebranding to StandaloneStreamChat.
set(MOD_ID "SimpleBSStreamChat")

cmake_minimum_required(VERSION 3.22)
project(${COMPILE_ID})

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CMAKE_INTERPROCEDURAL_OPTIMIZATION TRUE)
set(CMAKE_EXPORT_COMPILE_COMMANDS TRUE)

set(PROJECT_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)
set(SOURCE_DIR ${PROJECT_ROOT}/src)
set(INCLUDE_DIR ${PROJECT_ROOT}/include)

# stop symbol leaking
add_link_options(-Wl,--exclude-libs,ALL)

add_compile_options(-frtti -fPIE -fPIC -fexceptions -fdeclspec -fvisibility=hidden -Wno-c23-extensions -Wno-vla-cxx-extension)

add_compile_definitions(VERSION="${MOD_VERSION}")
add_compile_definitions(MOD_ID="${MOD_ID}")
add_compile_definitions(SBSC_BS_1_40)

# Trim source folder in logcat messages
string(LENGTH "${PROJECT_ROOT}/" FOLDER_LENGTH)
add_compile_definitions(PAPER_ROOT_FOLDER_LENGTH=${FOLDER_LENGTH})

RECURSE_FILES(cpp_file_list ${SOURCE_DIR}/*.cpp)
RECURSE_FILES(c_file_list ${SOURCE_DIR}/*.c)

# beatsaber-hook 6 predates flamingo: each mod compiles its inline hooker in.
RECURSE_FILES(inline_hook_c ${EXTERN_DIR}/includes/beatsaber-hook/shared/inline-hook/*.c)
RECURSE_FILES(inline_hook_cpp ${EXTERN_DIR}/includes/beatsaber-hook/shared/inline-hook/*.cpp)

add_library(
    ${COMPILE_ID}
    SHARED
    ${cpp_file_list}
    ${c_file_list}
    ${inline_hook_c}
    ${inline_hook_cpp}
)

target_include_directories(${COMPILE_ID} PRIVATE ${INCLUDE_DIR})
target_include_directories(${COMPILE_ID} PUBLIC ${SHARED_DIR})

# The system zlib (an NDK stable library) unpacks requested map zips; the
# inline hooker logs through liblog.
target_link_libraries(${COMPILE_ID} PRIVATE -lz -llog)

# Button and icon PNGs, compiled in (see ../cmake/EmbedAssets.cmake).
include(${PROJECT_ROOT}/cmake/EmbedAssets.cmake)
embed_assets(${COMPILE_ID} ${PROJECT_ROOT}/assets)

# qpm dependencies (libs + includes), generated by `qpm restore`
include(extern.cmake)

# Ship a stripped .so; keep the unstripped one in build/debug for ndk-stack.
add_custom_command(
    TARGET ${COMPILE_ID}
    POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "debug"
    COMMAND ${CMAKE_COMMAND} -E rename "lib${COMPILE_ID}.so" "debug/lib${COMPILE_ID}.so"
    COMMAND ${CMAKE_STRIP} -d --strip-all "debug/lib${COMPILE_ID}.so" -o "lib${COMPILE_ID}.so"
    COMMENT "Create stripped and debug libraries"
)
